    id = voctroubles.12
    title = voctroubles.12.t
    desc = voctroubles.12.d
    picture = GFX_report_event_generic_sign_treaty1

    is_triggered_only = yes

    trigger = {
        original_tag = YAN
    }

    option = {
        name = voctroubles.12.a
        ai_chance = { factor = 1 }
        available = {
            NOT = {
                has_global_flag = South_Zone_Crisis_Resolved
            }
        }

        custom_effect_tooltip = VOC_CW_Start_Tooltip_1_tt

        hidden_effect = {
            country_event = { id = beakavian_emergency.1 days = 5 }

            if = {
                limit = { country_flag = voc_mandate_harmony_entrenched_flag }
                country_event = { id = beakavian_emergency_debug.1 }
            }
            if = {
                limit = { country_flag = voc_mandate_supremacy_entrenched_flag }
                country_event = { id = beakavian_emergency_debug.2 }
            }
            country_event = { id = beakavian_emergency_debug.3 }
        }

        end_puppet = VOC

        1153 = { add_core_of = HAF }
        1154 = { add_core_of = HAF }
        1152 = { add_core_of = THA }
        1190 = { add_core_of = THA }
        1143 = { add_core_of = HAF }
        1142 = { add_core_of = NKO }
        1141 = { add_core_of = NKO }
        1192 = { add_core_of = TAO }
        1146 = { add_core_of = TAO }
        1144 = { add_core_of = THA }
        1150 = { add_core_of = XHO }
        1145 = { add_core_of = XHO }
        1148 = { add_core_of = XHO }
        1155 = { add_core_of = WAD }
        1149 = { add_core_of = VOC }

        UBE = {
            release = HAF
            release = IDO
            release = IST
            release = NKO
            release = TAO
            release = THA
            release = XHO
            release = WAD
        }

        if = {
            limit = {
                1149 = { is_owned_by = UBE }
            }
            1149 = { transfer_state_to = VOC }
        }

        every_country = {
            limit = {
                OR = {
                    tag = HAF
                    tag = IDO
                    tag = IST
                    tag = NKO
                    tag = TAO
                    tag = THA
                    tag = XHO
                    tag = WAD
                    tag = VOC
                }
                exists = yes
            }

            every_other_country = {
                limit = {
                    OR = {
                        tag = HAF
                        tag = IDO
                        tag = IST
                        tag = NKO
                        tag = TAO
                        tag = THA
                        tag = XHO
                        tag = WAD
                        tag = VOC
                    }
                    exists = yes
                }

                declare_war_on = {
                    target = PREV
                    type = annex_everything
                }
                add_civil_war_target = PREV
            }
        }

        UBE = {
            if = {
                limit = { has_character = various_warlords }
                retire_character = various_warlords
            }
            if = {
                limit = { has_character = UBE_Saviour_of_the_South }
                promote_character = UBE_Saviour_of_the_South
            }
            set_politics = {
                ruling_party = democratic
                elections_allowed = no
            }
            set_popularities = {
                democratic = 60
                fascism = 30
                neutrality = 10
                communism = 0
            }
        }
    }
    option = {
        name = voctroubles.12.b
        ai_chance = {
            factor = 1
        }
        available = {
            has_global_flag = South_Zone_Crisis_Resolved
        }
    }
}

    immediate = {
        clamp_variable = {
            var = South_Zone_Corruption_Investigation

        }

    option = { # Normal Option available if nothing else
        name = South_Zone_Crisis.9.a

                    AND = {
                        check_variable = { South_Zone_Merc_Strength_Crisis > 4 }
                        check_variable = { South_Zone_Zebra_Tribes_Crisis < 1 }
                    }
                    AND = {
                        check_variable = { South_Zone_Zebra_Tribes_Crisis > 4 }
                        check_variable = { South_Zone_Merc_Strength_Crisis < 1 }
                    }
                    AND = {
                        check_variable = { South_Zone_Merc_Strength_Crisis = 0 }
                        check_variable = { South_Zone_Zebra_Tribes_Crisis = 0 }

                    AND = {
                        check_variable = { South_Zone_Crisis_Air_Supply > 2 }
                        check_variable = { South_Zone_Helicopter_Investment = 3 }
                        check_variable = { South_Zone_Merc_Strength_Crisis > 4 }
                        check_variable = { South_Zone_Zebra_Tribes_Crisis < 1 }
                        check_variable = { South_Zone_Tourism_Crisis > 4 }
                    }
                }
            }
        }
        custom_effect_tooltip = South_Zone_Event_9_Effect_A_tt
        hidden_effect = {
            set_country_flag = South_Zone_Corruption_Option_A_Taken

country_event = { # Event Hub - This event should be able to dynamically decide what events the player will get served to them. Every 180 days it gets repinged. And it works by using the country flag from the decision to determine the chance.
    id = South_Zone_Crisis_Corruption.2
    title = South_Zone_Crisis_Corruption.2.t
    desc = South_Zone_Crisis_Corruption.2.desc
    picture = GFX_report_event_generic
    is_triggered_only = yes
    hidden = yes
    trigger = {
        NOT = {
            has_country_flag = South_Zone_Corruption_Hub_Ended
        }
    }
    immediate = {
        IF = {
            limit = {
                NOT = {
                    has_country_flag = South_Zone_Corruption_Chain_Completed
                }
            }
            country_event = { id = South_Zone_Crisis_Corruption.2 days = 180 }
        }
        ELSE = {
            limit = {
                has_country_flag = South_Zone_Corruption_Chain_Completed
            }
            set_country_flag = South_Zone_Corruption_Hub_Ended
        }
    }
    option = {
        name = South_Zone_Crisis_Corruption.2.a
        trigger = {
            has_country_flag = South_Zone_Corruption_Option_A_Taken
        }
        country_event = { id = South_Zone_Tasty.1 days = 14 }
    }
    option = {
        name = South_Zone_Crisis_Corruption.2.b
        trigger = {
            has_country_flag = South_Zone_Corruption_Option_B_Taken
        }
        country_event = { id = South_Zone_Tasty.1 days = 14 }
        random_list = {
            50 = {
                country_event = { id = South_Zone_Tasty.2 days = 54 }
            }
            50 = {
                country_event = { id = South_Zone_Tasty.3 days = 71 }
            }
        }
    }
    option = {
        name = South_Zone_Crisis_Corruption.2.c
        trigger = {
            has_country_flag = South_Zone_Corruption_Option_C_Taken
        }
		country_event = { id = South_Zone_Tasty.1 days = 8 }
		country_event = { id = South_Zone_Tasty.4 days = 33 }
		random_list = {
			50 = {
				country_event = { id = South_Zone_Tasty.5 days = 87 }
			}
			50 = {
				country_event = { id = South_Zone_Tasty.6 days = 55 }
			}
		}
    }
    option = {
        name = South_Zone_Crisis_Corruption.2.d
        trigger = {
            has_country_flag = South_Zone_Corruption_Option_D_Taken
        }
		country_event = { id = South_Zone_Tasty.1 days = 6 }
		country_event = { id = South_Zone_Tasty.7 days = 44 }
		country_event = { id = South_Zone_Tasty.8 days = 22 }
		random_list = {
			50 = {
				country_event = { id = South_Zone_Tasty.5 days = 56 }
			}
			50 = {
				country_event = { id = South_Zone_Tasty.5 days = 111 }
			}
		}
    }
    option = {
        name = South_Zone_Crisis_Corruption.2.e
        trigger = {
            has_country_flag = South_Zone_Corruption_Option_E_Taken
        }
		country_event = { id = South_Zone_Tasty.1 days = 4 }
		country_event = { id = South_Zone_Tasty.9 days = 35 }
		country_event = { id = South_Zone_Tasty.10 days = 61 }
		country_event = { id = South_Zone_Tasty.11 days = 101 }
		country_event = { id = South_Zone_Tasty.12 days = 137 }	
    }
    option = {
        name = South_Zone_Crisis_Corruption.2.f
        trigger = {
            has_country_flag = South_Zone_Corruption_Option_F_Taken
        }
		country_event = { id = South_Zone_Tasty.1 days = 2 }
		country_event = { id = South_Zone_Tasty.9 days = 35 }
		country_event = { id = South_Zone_Tasty.10 days = 61 }
		country_event = { id = South_Zone_Tasty.11 days = 101 }
		country_event = { id = South_Zone_Tasty.12 days = 137 }
		country_event = { id = South_Zone_Tasty.13 days = 15 }	
		country_event = { id = South_Zone_Tasty.14 days = 22 }
		country_event = { id = South_Zone_Tasty.15 days = 92 }
    }
}

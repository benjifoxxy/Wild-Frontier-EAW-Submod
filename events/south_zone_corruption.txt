country_event = { # fires when corruption reaches 35% (34) investigated, caps variable now at 30 minimum
    id = South_Zone_Shawarma.1
    title = South_Zone_Shawarma.1.t
    desc = South_Zone_Shawarma.1.desc
    picture = GFX_report_event_generic
    is_triggered_only = yes
    trigger = {
        check_variable = {
            var = South_Zone_Corruption_Investigation > 34
        }
    }
    immediate = {
        clamp_variable = {
            var = South_Zone_Corruption_Investigation
            min = 30
            max = 100
        }
    }
    option = {
        name = South_Zone_Shawarma.1.a
    }
}
